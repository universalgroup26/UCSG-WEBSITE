   117→function ScrollReveal({ children, className = '', delay = 0 }: { children: React.ReactNode; className?: string; delay?: number }) {
   118→  const ref = useRef(null);
   119→  const isInView = useInView(ref, { once: true, margin: '-60px' });
   120→  return (
   121→    <motion.div
   122→      ref={ref}
   123→      className={className}
   124→      initial={{ opacity: 0, y: 24 }}
   125→      animate={isInView ? { opacity: 1, y: 0 } : {}}
   126→      transition={{ delay, duration: 0.5, ease: [0.22, 1, 0.36, 1] }}
   127→    >
   128→      {children}
   129→    </motion.div>
   130→  );
   131→}
