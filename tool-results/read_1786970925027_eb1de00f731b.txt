   116→function ScrollReveal({ children, className = '', delay = 0 }: { children: React.ReactNode; className?: string; delay?: number }) {
   117→  const ref = useRef(null);
   118→  const isInView = useInView(ref, { once: true, margin: '-60px' });
   119→  return (
   120→    <motion.div
   121→      ref={ref}
   122→      className={className}
   123→      initial={{ opacity: 0, y: 24 }}
   124→      animate={isInView ? { opacity: 1, y: 0 } : {}}

   126→    >
   127→      {children}
   128→    </motion.div>
   129→  );
   130→}
